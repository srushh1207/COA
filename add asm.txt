.MODEL SMALL
.STACK 100H

.DATA
NUM1 DB 20
NUM2 DB 5

ADDITION DW ?
SUBTRACTION DB ?
MULTIPLICATION DW ?
QUOTIENT DB ?
REMAINDER DB ?

.CODE
MAIN PROC

    MOV AX, @DATA
    MOV DS, AX

    ; Addition
    MOV AL, NUM1
    MOV BL, NUM2
    ADD AL, BL
    MOV AH, 00H
    MOV ADDITION, AX

    ; Subtraction
    MOV AL, NUM1
    MOV BL, NUM2
    SUB AL, BL
    MOV SUBTRACTION, AL

    ; Multiplication
    MOV AL, NUM1
    MOV BL, NUM2
    MUL BL
    MOV MULTIPLICATION, AX

    ; Division
    MOV AL, NUM1
    MOV AH, 00H
    MOV BL, NUM2
    DIV BL
    MOV QUOTIENT, AL
    MOV REMAINDER, AH

    MOV AH, 4CH
    INT 21H

MAIN ENDP
END MAIN


.MODEL SMALL
.STACK 100H

.DATA
ARR DB 5, 2, 4, 1, 3
N DB 5

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX

    MOV CL, N
    DEC CL
    JZ EXIT

L1:
    MOV SI, 0
    MOV BL, CL

L2:
    MOV AL, ARR[SI]
    MOV DL, ARR[SI+1]

    CMP AL, DL
    JBE NEXT

    MOV ARR[SI], DL
    MOV ARR[SI+1], AL

NEXT:
    INC SI
    DEC BL
    JNZ L2

    DEC CL
    JNZ L1

EXIT:
    MOV AH, 4CH
    INT 21H

MAIN ENDP
END MAIN

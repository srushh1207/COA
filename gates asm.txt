
.MODEL SMALL
.STACK 100H

.DATA
NUM1 DB 20
NUM2 DB 5

AND_RESULT DB ?
OR_RESULT DB ?
XOR_RESULT DB ?
NOT_RESULT DB ?

.CODE
MAIN PROC

    MOV AX, @DATA
    MOV DS, AX

    ; AND
    MOV AL, NUM1
    MOV BL, NUM2
    AND AL, BL
    MOV AND_RESULT, AL

    ; OR
    MOV AL, NUM1
    MOV BL, NUM2
    OR AL, BL
    MOV OR_RESULT, AL

    ; XOR
    MOV AL, NUM1
    MOV BL, NUM2
    XOR AL, BL
    MOV XOR_RESULT, AL

    ; NOT
    MOV AL, NUM1
    NOT AL
    MOV NOT_RESULT, AL

    MOV AH, 4CH
    INT 21H

MAIN ENDP
END MAIN

.MODEL SMALL
.STACK 100H

.CODE
MAIN PROC
    MOV AH, 00H
    MOV AL, 12H
    INT 10H

    MOV AH, 0CH
    MOV AL, 0FH

    MOV CX, 200
    MOV DX, 100
    MOV SI, 100

TOP:
    INT 10H
    INC CX
    DEC SI
    JNZ TOP

    MOV CX, 200
    MOV DX, 200
    MOV SI, 100

BOTTOM:
    INT 10H
    INC CX
    DEC SI
    JNZ BOTTOM

    MOV CX, 200
    MOV DX, 100
    MOV SI, 100

LEFT:
    INT 10H
    INC DX
    DEC SI
    JNZ LEFT

    MOV CX, 300
    MOV DX, 100
    MOV SI, 100

RIGHT:
    INT 10H
    INC DX
    DEC SI
    JNZ RIGHT

    MOV AH, 00H
    INT 16H

    MOV AH, 00H
    MOV AL, 03H
    INT 10H

    MOV AH, 4CH
    INT 21H
MAIN ENDP
END MAIN
